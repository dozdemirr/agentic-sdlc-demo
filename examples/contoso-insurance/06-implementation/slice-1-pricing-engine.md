### Key Code — `PricingEndpoints.cs`

```csharp
public static class PricingEndpoints
{
    public static void MapPricingEndpoints(this WebApplication app)
    {
        var group = app.MapGroup("/api/pricing")
            .WithTags("Pricing")
            .RequireAuthorization();

        group.MapPost("/calculate", async (
            RiskProfile risk,
            IPremiumCalculator calculator,
            CancellationToken ct) =>
        {
            var result = await calculator.CalculateAsync(risk, ct);
            return Results.Ok(result);
        })
        .WithName("CalculatePremium")
        .Produces<PremiumResult>(200)
        .ProducesValidationProblem();

        group.MapGet("/health", () => Results.Ok(new { Status = "Healthy" }))
            .AllowAnonymous();
    }
}
```

### Key Code — `PremiumCalculator.cs` (migrated from legacy)

```csharp
public class PremiumCalculator : IPremiumCalculator
{
    private readonly IFactorTableRepository _factors;
    private readonly ILogger<PremiumCalculator> _logger;

    public PremiumCalculator(IFactorTableRepository factors, ILogger<PremiumCalculator> logger)
    {
        _factors = factors;
        _logger = logger;
    }

    public async Task<PremiumResult> CalculateAsync(RiskProfile risk, CancellationToken ct)
    {
        var basePremium = await _factors.GetBasePremiumAsync(risk.ProductType, ct);
        var areaFactor = await _factors.GetAreaFactorAsync(risk.PostCode, ct);
        var vehicleFactor = await _factors.GetVehicleGroupFactorAsync(risk.VehicleAbiCode, ct);
        var ageFactor = await _factors.GetAgeFactorAsync(risk.DriverAge, ct);
        var ncdDiscount = await _factors.GetNcdDiscountAsync(risk.NcdYears, ct);

        var grossPremium = basePremium * areaFactor * vehicleFactor * ageFactor;
        var netPremium = grossPremium * ncdDiscount;
        var ipt = netPremium * 0.12m; // 12% IPT

        _logger.LogInformation("Calculated premium for {PostCode}, age {Age}: {Premium}",
            risk.PostCode, risk.DriverAge, netPremium);

        return new PremiumResult
        {
            BasePremium = basePremium,
            Loadings = new Dictionary<string, decimal>
            {
                ["AreaRating"] = areaFactor,
                ["VehicleGroup"] = vehicleFactor,
                ["AgeRating"] = ageFactor
            },
            Discounts = new Dictionary<string, decimal>
            {
                ["NCD"] = ncdDiscount
            },
            FinalPremium = Math.Round(netPremium, 2),
            IPT = Math.Round(ipt, 2),
            TotalPayable = Math.Round(netPremium + ipt, 2)
        };
    }
}
```

## PR 3/4 — Container + Deployment (156 LOC)

### Dockerfile

```dockerfile
FROM mcr.microsoft.com/dotnet/aspnet:8.0 AS base
WORKDIR /app
EXPOSE 8080

FROM mcr.microsoft.com/dotnet/sdk:8.0 AS build
WORKDIR /src
COPY ["Contoso.Pricing.Api/Contoso.Pricing.Api.csproj", "Contoso.Pricing.Api/"]
COPY ["Contoso.Pricing.Core/Contoso.Pricing.Core.csproj", "Contoso.Pricing.Core/"]
COPY ["Contoso.Pricing.Data/Contoso.Pricing.Data.csproj", "Contoso.Pricing.Data/"]
RUN dotnet restore "Contoso.Pricing.Api/Contoso.Pricing.Api.csproj"
COPY . .
RUN dotnet publish "Contoso.Pricing.Api/Contoso.Pricing.Api.csproj" -c Release -o /app/publish

FROM base AS final
WORKDIR /app
COPY --from=build /app/publish .
ENTRYPOINT ["dotnet", "Contoso.Pricing.Api.dll"]
```

## PR 4/4 — Traffic Cutover + Legacy Decommission (89 LOC)

### Traffic Split Strategy

| Day | New Service | Legacy WCF | Validation |
|---|---|---|---|
| Day 1 | 0% (shadow only) | 100% | Compare responses — log discrepancies |
| Day 3 | 10% | 90% | Monitor error rate + latency in App Insights |
| Day 5 | 50% | 50% | Confirm premium parity in production |
| Day 7 | 100% | 0% (standby) | Full cutover — legacy remains deployable |
| Day 14 | 100% | Decommissioned | Remove WCF project from solution |

### Rollback Procedure

```powershell
# Emergency rollback — revert API Management to legacy endpoint
az apim api update \
  --resource-group rg-contoso-prod \
  --service-name apim-contoso \
  --api-id pricing-api \
  --service-url "https://legacy-vm.contoso.internal/RatingEngine.svc"
```

**Rollback time**: <2 minutes (API Management route change, no redeployment needed).
